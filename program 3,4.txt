program 3:
select SYSDATE from dual;
SELECT EMPNO,ENAME,SAL, ROUND(sal * 1.15, 0) "New Salary" FROM emp;
SELECT ENAME,INITCAP(ENAME),LENGTH(ENAME) from EMP;
SELECT ENAME,HIREDATE,TRUNC(MONTHS_BETWEEN(SYSDATE,HIREDATE))
"MONTHS_WORKED" FROM EMP ORDER BY
MONTHS_BETWEEN(SYSDATE,HIREDATE);

SELECT MIN(SAL),MAX(SAL),AVG(SAL),SUM(SAL) FROM EMP;

SELECT SUM(SAL) FROM EMP WHERE DEPTNO=20 AND JOB='ANALYST';
SELECT DEPTNO,COUNT(*) FROM EMP
GROUP BY DEPTNO;
SELECT JOB,COUNT(*) FROM EMP
GROUP BY JOB;
SELECT JOB,MIN(SAL), FROM EMP
GROUP BY JOB;
SELECT JOB,MIN(SAL),MAX(SAL),AVG(SAL),SUM(SAL) FROM EMP

SELECT DEPTNO,JOB,COUNT(*) FROM EMP
GROUP BY DEPTNO,JOB;
SELECT DEPTNO,JOB,COUNT(*) FROM EMP
